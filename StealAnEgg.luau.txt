local function optimize(obj)
    pcall(function()
        if obj:IsA("ParticleEmitter")
        or obj:IsA("Trail")
        or obj:IsA("Smoke")
        or obj:IsA("Fire")
        or obj:IsA("Sparkles") then
            obj.Enabled = false
        elseif obj:IsA("PostEffect") then
            obj.Enabled = false
        elseif obj:IsA("BasePart") then
            obj.Material = Enum.Material.SmoothPlastic
            obj.CastShadow = false
        end
    end)
end

for _,v in ipairs(workspace:GetDescendants()) do
    optimize(v)
end

workspace.DescendantAdded:Connect(optimize)

--// Teleport từng 50 studs, delay 0.8s
local function smoothTeleport()
    if teleporting or not savedCFrame then return end

    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    teleporting = true

    while teleporting and root.Parent do
        local target = savedCFrame.Position
        local current = root.Position
        local offset = target - current
        local distance = offset.Magnitude

        if distance <= 3 then
            root.CFrame = savedCFrame
            break
        end

        local direction = offset.Unit
        local step = math.min(50,distance)
        local nextPos = current + direction * step

        root.CFrame = CFrame.new(nextPos) * savedCFrame.Rotation

        task.wait(0.8)
    end

    teleporting = false
end

tpButton.MouseButton1Click:Connect(function()
    task.spawn(smoothTeleport)
end)

    minimized = not minimized

    for _,v in ipairs(main:GetChildren()) do
        if v ~= header and v ~= stroke and v ~= corner then
            v.Visible = not minimized
        end
    end

    if minimized then
        main.Size = UDim2.fromOffset(260,38)
        minus.Text = "+"
    else
        main.Size = UDim2.fromOffset(260,240)
        minus.Text = "−"
    end
end)

--// Drag mobile + PC
local dragging = false
local dragStart
local startPos

header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPos = main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

header.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
    or input.UserInputType == Enum.UserInputType.Touch then
        dragStart = dragStart or input.Position
    end
end)

RunService.RenderStepped:Connect(function()
    if dragging and dragStart then
        local mouse = player:GetMouse()
        local delta = Vector2.new(mouse.X,mouse.Y) -
            Vector2.new(dragStart.X,dragStart.Y)

        main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)
